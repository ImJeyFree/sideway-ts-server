cmake_minimum_required(VERSION 3.20)
project(sideway-ts-server LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT WIN32)
 message(FATAL_ERROR "Windows 전용 서버입니다")
endif()
if(MSVC)
 add_compile_options(/utf-8 /W3)
 add_compile_definitions(_CRT_SECURE_NO_WARNINGS WIN32_LEAN_AND_MEAN)
endif()
set(TUNER_CORE_SDK "${CMAKE_CURRENT_SOURCE_DIR}/sdk" CACHE PATH "튜너 코어 SDK 경로")
set(PUBLIC_CORE src/TsBroadcaster.cpp src/TunerClient.cpp src/ChannelStore.cpp src/HttpStreamer.cpp src/UdpStreamer.cpp src/RtpStreamer.cpp)
add_executable(sideway-ts-server ${PUBLIC_CORE} src/TrayApp.cpp src/main.cpp)
target_include_directories(sideway-ts-server PRIVATE include sdk)
target_link_libraries(sideway-ts-server PRIVATE ws2_32 ole32 shell32 user32 gdi32)
if(EXISTS "${TUNER_CORE_SDK}/bin/SidewayTunerCore.dll")
 add_custom_command(TARGET sideway-ts-server POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
 "${TUNER_CORE_SDK}/bin/SidewayTunerCore.dll" "$<TARGET_FILE_DIR:sideway-ts-server>/SidewayTunerCore.dll")
else()
 message(STATUS "SDK DLL이 없습니다. 서버 빌드는 가능하지만 실행하려면 공식 DLL이 필요합니다.")
endif()
include(CTest)
if(BUILD_TESTING)
 add_library(mock-tuner-core SHARED tests/MockTunerCore.cpp)
 target_compile_definitions(mock-tuner-core PRIVATE STC_BUILD)
 target_include_directories(mock-tuner-core PRIVATE include sdk)
 set_target_properties(mock-tuner-core PROPERTIES OUTPUT_NAME SidewayTunerCore RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/tests" ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/tests")
 add_executable(stream-tests tests/StreamTests.cpp ${PUBLIC_CORE})
 target_include_directories(stream-tests PRIVATE include sdk)
 target_link_libraries(stream-tests PRIVATE ws2_32)
 set_target_properties(stream-tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/tests")
 add_dependencies(stream-tests mock-tuner-core)
 add_test(NAME stream-regressions COMMAND stream-tests)
 set_tests_properties(stream-regressions PROPERTIES TIMEOUT 30)
 add_library(wrong-version-core SHARED tests/MockTunerCore.cpp)
 target_compile_definitions(wrong-version-core PRIVATE STC_BUILD MOCK_API_VERSION=999)
 target_include_directories(wrong-version-core PRIVATE include sdk)
 set_target_properties(wrong-version-core PROPERTIES OUTPUT_NAME SidewayTunerCore RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/wrong-version" ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/wrong-version")
 foreach(mode IN ITEMS missing version)
  add_executable(loader-${mode} tests/LoaderTests.cpp src/TunerClient.cpp src/TsBroadcaster.cpp)
  target_include_directories(loader-${mode} PRIVATE include sdk)
  if(mode STREQUAL "version")
   set_target_properties(loader-${mode} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/wrong-version")
   add_dependencies(loader-${mode} wrong-version-core)
  else()
   set_target_properties(loader-${mode} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/missing-dll")
  endif()
  add_test(NAME loader-${mode} COMMAND loader-${mode} ${mode})
 endforeach()
 find_program(NODE_EXECUTABLE node)
 if(NODE_EXECUTABLE)
  add_test(NAME dashboard-regressions COMMAND "${NODE_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/DashboardTests.js")
 endif()
endif()
